# all :所有
```
# 如何显示工资比部门30的员工工资高的的员工的姓名,工资和部门号
SELECT ename, sal,deptno
	FROM emp
	WHERE sal > (
		SELECT MAX(sal) 
			FROM emp
			WHERE deptno = 30
		);
```
# any : 其中一个
```
# 如何显示工资比部门30的其中一个员工工资高的的员工的姓名,工资和部门号
SELECT ename, sal,deptno
	FROM emp
	WHERE sal > ANY(
		SELECT sal 
			FROM emp
			WHERE deptno = 30
		);
# 或
SELECT ename, sal,deptno
	FROM emp
	WHERE sal > (
		SELECT MIN(sal)
			FROM emp
			WHERE deptno = 30
		);
		
		
SELECT * FROM emp;
```
